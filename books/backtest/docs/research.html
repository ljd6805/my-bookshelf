import math

def mulberry32(seed):
    a = seed & 0xFFFFFFFF
    def rnd():
        nonlocal a
        a = (a + 0x6D2B79F5) & 0xFFFFFFFF
        t = a
        t = ((t ^ (t >> 15)) * (t | 1)) & 0xFFFFFFFF
        t ^= (t + (((t ^ (t >> 7)) * (t | 61)) & 0xFFFFFFFF)) & 0xFFFFFFFF
        return ((t ^ (t >> 14)) & 0xFFFFFFFF) / 4294967296
    return rnd

def gauss(rnd):
    u = 1 - rnd(); v = rnd()
    return math.sqrt(-2 * math.log(u)) * math.cos(2 * math.pi * v)

def tick(p):

def make_path(seed=13, days=750, per=8, kappa=0.006, sig=0.0058, gap=0.006, mu_of=None, start=20000, ex_days=(240, 490, 740), sig_of=None):
    rnd = mulberry32(seed); x = math.log(start); out = []
    for d in range(days):
        mu = math.log(mu_of(d) if mu_of else 20000)
        if d > 0:
            x += gap * gauss(rnd)
            if d in ex_days:
                x = math.log(math.exp(x) - DIV)
        row = []
        for k in range(per):
            if k > 0 or d == 0:
                pass
            if k > 0:
                x += kappa * (mu - x) + (sig_of(d) if sig_of else sig) * gauss(rnd)
            row.append(rtick(math.exp(x)))
        out.append(row)
    return out

class Book:
    def __init__(s, levels, qty, capital, cost=True, through=False):

    def buy(s, i, p, t):
        a = p * s.q; f = FEE(a) if s.cost else 0
        s.armed[i] = False
        if a + f > s.cash: return
        s.cash -= a + f; s.fees += f; s.lots[i] = a + f; s.fills.append((t, 'B', i, p))
    def sell(s, i, p, t, how='limit'):
        a = p * s.q; f = FEE(a) if s.cost else 0; x = TAX(a) if s.cost else 0
        s.cash += a - f - x; s.fees += f; s.taxes += x; s.realized += a - f - x - s.lots[i]; s.lots[i] = None
        s.fills.append((t, 'S', i, p))

        if i in tod: return True
        tod.add(i); return False
    def shares(s): return sum(1 for l in s.lots if l) * s.q
    def move(s, a, b, t, gap):

        if b < a:
            for i in range(len(s.lots) - 1, -1, -1):
                L = s.L[i]; hit = b < L if s.through else b <= L
                if s.lots[i] is None and s.armed[i] and hit and not s.used(i): s.buy(i, min(L, b) if gap else L, t)
        else:
            for i in range(len(s.lots)):
                U = s.L[i + 1]; hit = b > U if s.through else b >= U
                if s.lots[i] is not None and hit and not s.used(i): s.sell(i, max(U, b) if gap else U, t)
        for i in range(len(s.lots)):
            if s.lots[i] is None and b > s.L[i]: s.armed[i] = True

            if s.lots[i] is not None: s.sell(i, p, t, 'stop')

def run(days, levels=None, qty=5, capital=3_000_000, cost=True, through=False, stop=None, resume=None,
        lookahead=False, ex_days=(), div_cash=True, mode='path', start=0, record=False):

    for d, row in enumerate(days):

        if mode in ('optimistic', 'onefill'): pts = [row[0], row[2], row[1], row[3]]
        elif mode == 'pessimistic': pts = [row[0], row[1], row[2], row[3]]
        else: pts = row
        b.today = set() if mode == 'onefill' else None
        if d in ex_days and div_cash:
            dv = b.shares() * DIV; b.cash += dv; b.divs += dv
        if prev is None:
            for i in range(len(b.lots)):

        if b.paused and resume is not None:

            if ref >= resume:
                b.paused = False
                for i in range(len(b.lots)): b.armed[i] = pts[0] > levels[i]
        if lookahead and stop is not None and not b.paused and min(pts) <= stop:
            b.liquidate(pts[0], t)       # 버그: 오늘 저가를 보고 오늘 시가에 판다
        seq = ([prev] if prev is not None else []) + pts
        for k in range(1, len(seq)):
            a, c = seq[k - 1], seq[k]; gap = (prev is not None and k == 1)
            if stop is not None and not b.paused and c <= stop < a:

            else:

        prev = pts[-1]
        curve.append(b.cash + b.shares() * prev)
